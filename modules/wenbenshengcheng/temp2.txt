def export_zhuiqiao_calculation_doc(product_id, save_dir, db_config=None, project_root=None):
    """
    独立生成“锥壳”Word计算书。

    功能：
    1. 查 产品设计活动表_元件计算结果表 中当前产品ID是否存在 元件名称='锥壳'
    2. 若不存在，则不输出Word
    3. 若存在，则根据 Name='锥壳名称' 的 Value 选择模板：
       - 卧式设备板式锥壳 -> 卧式设备板式锥壳-计算书模板.docx
       - 顶部板式锥壳     -> 顶部板式锥壳-计算书模板.docx
    4. 模板占位符支持：
       - 材料类型（锥壳）
       - 材料类型（壳体圆筒）
       - 系揽环板厚度（锥壳）（隐）
    5. 替换规则：
       - 括号外 = Name
       - 第一层括号内 = 元件名称
       - （隐）缺失时替换为 "-"
    6. 返回生成后的 docx 路径；若无锥壳则返回 None
    """

    if db_config is None:
        db_config = {
            "host": "localhost",
            "user": "root",
            "password": "123456",
            "database": "产品设计活动库",
            "charset": "utf8mb4",
            "cursorclass": pymysql.cursors.DictCursor,
        }

    save_dir = Path(save_dir)
    save_dir.mkdir(parents=True, exist_ok=True)

    if project_root is None:
        project_root = _find_project_root_with_templates(Path(__file__).resolve())
    else:
        project_root = Path(project_root).resolve()

    template_dir = project_root / "templates" / "calculation_docs" / "锥壳"
    if not template_dir.exists():
        raise FileNotFoundError(f"锥壳模板目录不存在：{template_dir}")

    conn = None
    try:
        conn = pymysql.connect(**db_config)
        cursor = conn.cursor()

        # 一次性读取当前产品ID所有元件计算结果
        cursor.execute(
            """
            SELECT 元件名称, Name, Value
            FROM 产品设计活动表_元件计算结果表
            WHERE 产品ID = %s
            """,
            (product_id,),
        )
        rows = cursor.fetchall()

        if not rows:
            print(f"[锥壳计算书] 产品ID={product_id} 未找到任何元件计算结果，不生成计算书。")
            return None

        # 构造成：component_map[元件名称][Name] = Value
        component_map = {}
        for row in rows:
            component_name = str(row.get("元件名称") or "").strip()
            name = str(row.get("Name") or "").strip()
            value = "" if row.get("Value") is None else str(row.get("Value"))

            if not component_name or not name:
                continue

            component_map.setdefault(component_name, {})
            component_map[component_name][name] = value

        # 必须存在锥壳元件，否则不生成锥壳计算书
        if "锥壳" not in component_map:
            print(f"[锥壳计算书] 产品ID={product_id} 不存在元件名称“锥壳”，不生成计算书。")
            return None

        lug_type = str(component_map["锥壳"].get("锥壳名称", "")).strip()
        if not lug_type:
            raise ValueError(f"产品ID={product_id} 的锥壳记录中缺少 Name='锥壳名称' 对应的 Value。")

        template_name_map = {
            "卧式设备板式锥壳": "卧式设备板式锥壳-计算书模板.docx",
            "顶部板式锥壳": "顶部板式锥壳-计算书模板.docx",
        }

        template_filename = template_name_map.get(lug_type)
        if not template_filename:
            raise ValueError(f"暂不支持的锥壳名称类型：{lug_type}")

        template_path = template_dir / template_filename
        if not template_path.exists():
            raise FileNotFoundError(f"模板不存在：{template_path}")

        doc = Document(str(template_path))

        for p in _iter_all_paragraphs(doc):
            _replace_placeholders_in_paragraph_by_component_map(
                paragraph=p,
                component_map=component_map,
            )

        timestamp = datetime.datetime.now().strftime("%Y%m%d_%H%M%S")
        out_name = f"锥壳计算书_{lug_type}_{timestamp}.docx"
        output_path = save_dir / out_name
        doc.save(str(output_path))

        print(f"[锥壳计算书] 生成成功：{output_path}")
        return str(output_path)

    finally:
        if conn:
            conn.close()
